    // FileHandler.h / FileHandler.cpp
    class FileHandler {

    5) 성공 시 `return true;`
    6) **DRY:** CsvExporter와 CSV 본문 생성 로직 중복 최소화

       · 또는 FileHandler가 CsvExporter::exportFilteredFeedbacks() 결과를 ofs에 write

bool FileHandler::saveToCsv(const std::vector<Feedback>& data, const std::string& path) {
    std::ofstream ofs(path, std::ios::binary);

    return true;
}
